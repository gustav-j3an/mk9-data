#### Etapa 1: Preflight, Backup Independente e Criação do Workspace Reconciliado

```bash
# 1. Validar projeto alvo com o guard de seguranca
TARGET_PROJECT="tafjceemuetqjyngresg"
node scripts/guard-homolog-env.mjs "$TARGET_PROJECT"

# 2. Criar diretório temporário isolado fora do repositório
WORKDIR="$(mktemp -d)"
echo "Criado workdir temporário: $WORKDIR"

# 3. Inicializar projeto temporário Supabase
supabase init --force --workdir "$WORKDIR"

# 4. Baixar o histórico canônico das 47 migrations já aplicadas na homologação
#    (Atua como backup independente e canônico do schema atual)
supabase migration fetch --remote="$TARGET_PROJECT" --workdir "$WORKDIR"

# 5. Validar que exatamente 47 arquivos foram recuperados e que os timestamps são únicos
fetched_count="$(ls "$WORKDIR/supabase/migrations"/*.sql | wc -l)"
unique_versions="$(ls "$WORKDIR/supabase/migrations"/*.sql | xargs -n1 basename | cut -d_ -f1 | sort -u | wc -l)"

if [ "$fetched_count" -ne 47 ] || [ "$unique_versions" -ne 47 ]; then
  echo "ERRO CRÍTICO: Esperado exatamente 47 migrations únicas, encontrado $fetched_count (únicas: $unique_versions)!" >&2
  rm -rf "$WORKDIR"
  exit 1
fi
echo "Histórico canônico de 47 migrations validado com sucesso no workspace isolado."

# 6. Conferir status de paridade antes de iniciar a aplicação
supabase migration list --remote="$TARGET_PROJECT" --workdir "$WORKDIR"
```

#### Etapa 2: Aplicação Controlada com Stop-Gate em Cada Migration

```bash
migrations=(
  supabase/migrations/20261105165500_m30_harden_industry_active_checkin.sql
  supabase/migrations/20261106131614_m30_serialize_industry_checkin.sql
  supabase/migrations/20261106140000_m37_pending_visit_active_route_guard.sql
  supabase/migrations/20261106150000_m37_checkin_route_and_conflict_guard.sql
  supabase/migrations/20261107140000_m37_reconcile_materialized_pending_recovery.sql
)

for migration in "${migrations[@]}"; do
  mig_basename="$(basename "$migration")"
  expected_ver="$(echo "$mig_basename" | cut -d_ -f1)"
  echo "=========================================================="
  echo ">>> APLICANDO MIGRATION: $mig_basename"
  echo "=========================================================="

  # 1. Copiar migration para o workspace temporário
  cp "$migration" "$WORKDIR/supabase/migrations/"

  # 2. Executar via Supabase CLI com target explícito na homologação
  supabase migration up --remote="$TARGET_PROJECT" --workdir "$WORKDIR"

  # 3. Critério de Interrupção / Verificação de Parada
  #    Conferir via CLI se a versão consta aplicada
  echo ">>> Verificando registro da versão $expected_ver..."
  list_output="$(supabase migration list --remote="$TARGET_PROJECT" --workdir "$WORKDIR")"
  if ! echo "$list_output" | grep -q "$expected_ver"; then
    echo "ERRO FATAL: Migration $mig_basename não foi registrada no ledger remoto!" >&2
    echo "INTERROMPENDO O PROCESSO IMEDIATAMENTE PARA ANÁLISE." >&2
    exit 1
  fi

  echo ">>> Migration $mig_basename aplicada e verificada com sucesso."
done

# Descartar workspace temporário
rm -rf "$WORKDIR"
echo "Aplicação das 5 migrations concluída. Workspace temporário descartado."
```
